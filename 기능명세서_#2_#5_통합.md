### 3. 데이터베이스 쿼리

```sql
-- 챌린지 목록 조회 (필터 포함)
SELECT 
  c.challenge_id,
  c.platform,
  c.video_id,
  c.title,
  c.description,
  c.thumbnail_url,
  c.channel_name,
  c.channel_url,
  c.video_url,
  c.view_count,
  c.like_count,
  c.comment_count,
  c.share_count,
  c.upload_date,
  c.duration_seconds,
  c.difficulty,
  c.collected_at,
  c.is_verified,
  cat.category_id,
  cat.category_name,
  cat.category_code
FROM challenges c
LEFT JOIN categories cat ON c.primary_category_id = cat.category_id
WHERE c.is_active = TRUE
  AND (? IS NULL OR c.primary_category_id = ?)
  AND (? IS NULL OR c.platform = ?)
  AND (? IS NULL OR c.difficulty = ?)
ORDER BY 
  CASE WHEN ? = 'latest' THEN c.upload_date END DESC,
  CASE WHEN ? = 'popular' THEN c.view_count END DESC,
  CASE WHEN ? = 'views' THEN c.view_count END DESC
LIMIT ? OFFSET ?;
```

### 4. 비즈니스 로직

#### 4.1 관련 챌린지 추천

```javascript
function findRelatedChallenges(challenge, limit = 5) {
  // 같은 카테고리 + 유사 태그
  const relatedChallenges = challenges
    .filter(c => c.category_id === challenge.category_id)
    .filter(c => c.challenge_id !== challenge.challenge_id)
    .map(c => ({
      ...c,
      similarity_score: calculateSimilarity(challenge, c)
    }))
    .sort((a, b) => b.similarity_score - a.similarity_score)
    .slice(0, limit);
  
  return relatedChallenges;
}

function calculateSimilarity(challenge1, challenge2) {
  // 태그 겹침 / 전체 태그 수
  const commonTags = challenge1.tags.filter(t =>
    challenge2.tags.find(t2 => t2.tag_id === t.tag_id)
  );
  
  const allTags = new Set([
    ...challenge1.tags,
    ...challenge2.tags
  ]);
  
  return commonTags.length / allTags.size;
}
```

#### 4.2 참여도 계산

```javascript
function calculateEngagementRate(challenge) {
  // (좋아요 + 댓글 + 공유) / 조회수 * 100
  const total_interaction = 
    challenge.like_count + 
    challenge.comment_count + 
    challenge.share_count;
  
  if (challenge.view_count === 0) return 0;
  
  return (total_interaction / challenge.view_count * 100).toFixed(2);
}
```

### 3. 데이터베이스 쿼리

```sql
-- 트렌드 점수 계산 및 저장 (배치 작업)
INSERT INTO trends_daily (trend_date, category_id, top_10_challenges, avg_view_count, total_challenges_count, trend_score)
SELECT
  CURDATE(),
  cat.category_id,
  JSON_ARRAY(
    SELECT JSON_OBJECT(
      'challenge_id', c.challenge_id,
      'title', c.title,
      'view_count', c.view_count
    )
    FROM challenges c
    WHERE c.primary_category_id = cat.category_id
      AND DATE(c.upload_date) >= DATE_SUB(CURDATE(), INTERVAL 7 DAY)
    ORDER BY c.view_count DESC
    LIMIT 10
  ),
  AVG(c.view_count),
  COUNT(c.challenge_id),
  ROUND((AVG(c.view_count) / (SELECT MAX(view_count) FROM challenges)) * 100)
FROM categories cat
LEFT JOIN challenges c ON c.primary_category_id = cat.category_id
WHERE DATE(c.upload_date) >= DATE_SUB(CURDATE(), INTERVAL 7 DAY)
GROUP BY cat.category_id;
```

### 4. 비즈니스 로직

#### 4.1 트렌드 점수 계산

```javascript
function calculateTrendScore(challenges) {
  // 조회수, 좋아요, 댓글의 가중 평균
  // 가중치: 조회수 50%, 좋아요 30%, 댓글 20%
  
  const avgViews = challenges.reduce((sum, c) => sum + c.view_count, 0) / challenges.length;
  const avgLikes = challenges.reduce((sum, c) => sum + c.like_count, 0) / challenges.length;
  const avgComments = challenges.reduce((sum, c) => sum + c.comment_count, 0) / challenges.length;
  
  // 정규화 (0-100)
  const maxScore = 100;
  const score = (avgViews / 1000000 * 0.5 + avgLikes / 50000 * 0.3 + avgComments / 5000 * 0.2) * maxScore;
  
  return Math.min(100, score);
}
```

#### 4.2 트렌드 방향 결정

```javascript
function determineTrendDirection(currentScore, previousScore) {
  const percentChange = ((currentScore - previousScore) / previousScore) * 100;
  
  if (percentChange > 10) return 'UP';
  if (percentChange < -10) return 'DOWN';
  return 'STABLE';
}
```

#### 3.1 추천 알고리즘

```javascript
async function getRecommendations(limit = 10, days = 7) {
  // Step 1: 최근 N일 트렌드 데이터 조회
  const trends = await getTrendsData(days);
  
  // Step 2: 카테고리별 상위 챌린지 추출
  const categoryTopChallenges = {};
  trends.category_statistics.forEach(category => {
    const topChallenges = await getChallengesByCategory(
      category.category_id,
      5  // 카테고리당 상위 5개
    );
    categoryTopChallenges[category.category_id] = topChallenges;
  });
  
  // Step 3: 모든 챌린지를 하나의 배열로 병합
  const allChallenges = Object.values(categoryTopChallenges).flat();
  
  // Step 4: 트렌드 점수 기준으로 정렬
  const sorted = allChallenges.sort(
    (a, b) => b.trend_score - a.trend_score
  );
  
  // Step 5: 다양성 보장 (최대한 여러 카테고리)
  const diverseRecommendations = [];
  const categoryCount = {};
  const maxPerCategory = Math.ceil(limit / trends.category_statistics.length);
  
  for (const challenge of sorted) {
    if (diverseRecommendations.length >= limit) break;
    
    const categoryId = challenge.category.category_id;
    if (!categoryCount[categoryId]) {
      categoryCount[categoryId] = 0;
    }
    
    if (categoryCount[categoryId] < maxPerCategory) {
      diverseRecommendations.push(challenge);
      categoryCount[categoryId]++;
    }
  }
  
  return diverseRecommendations;
}
```
